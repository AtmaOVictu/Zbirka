JEWIK::ADAPTIVNI_TEMPO(tempo_dihanju):
    if tempo_dihanju < 0.2 Hz:
        STRETCH(sound, factor=2.0)  # počasno dihanje → počasnejši prehod
    elif tempo_dihanju > 0.4 Hz:
        COMPRESS(sound, factor=0.5)  # hitro dihanje → hitrejši obredni tok
    else:
        KEEP_ORIGINAL(sound)
